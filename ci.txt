name: CI/CD Pipeline

on:
 push:
 branches:
 - devops/try

jobs:
 build-and-secure:
 runs-on: ubuntu-latest

 steps:
 # 1. Check out the source code
 - name: Checkout Source Code
 uses: actions/checkout@v3
 with:
 fetch-depth: 0 # Required for Gitleaks to scan the full history

 # PHASE 3: SECURITY SCANS

 # 1. Detect hardcoded secrets
 - name: Gitleaks Scan
 uses: gitleaks/gitleaks-action@v2
 env:
 GITHUB_TOKEN: ${{ secrets.GITHUB_TOKEN }}

 # 2. SAST Scanning
 - name: Semgrep Scan
 uses: returntocorp/semgrep-action@v1
 with:
 config: "p/default"

 # 3. Dependency Scan
 - name: NPM Audit
 run: npm install && npm audit

 # PHASE 1 & 2: BUILD AND PUSH

 # Build the image locally first so Trivy can scan it
 - name: Build Docker Image Locally
 run: docker build -t my-secure-app:latest .

 # 4. Scan the Docker image
 - name: Trivy Image Scan
 uses: aquasecurity/trivy-action@master
 with:
 image-ref: "my-secure-app:latest"
 format: "table"
 exit-code: "1" # Fails the pipeline if critical issues are found
 ignore-unfixed: true
 severity: "CRITICAL,HIGH"

 # Log in to Docker Hub (Only happens if all security scans pass)
 - name: Log in to Docker Hub
 uses: docker/login-action@v2
 with:
 username: ${{ secrets.DOCKERHUB_USERNAME }}
 password: ${{ secrets.DOCKERHUB_TOKEN }}

 # Push the image to dockerhub
 - name: Push Docker Image
 uses: docker/build-push-action@v4
 with:
 context: .
 push: true
 tags: ${{ secrets.DOCKERHUB_USERNAME }}/my-secure-app:latest
